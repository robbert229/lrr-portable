# Reconstruction source-tree root (relative to the project root).
# Relocating the tree is one option + moving the directory — no build rewiring.
option('src_root', type : 'string', value : 'src',
       description : 'Directory containing the reconstruction source tree')

# Platform backend that satisfies the DX6-era API surface (finding 15):
#   native : miniwin (SDL3) — primary, end-goal build, host compiler, no cross file
#   winapi : MinGW-w64 i686 vs real DirectX headers — reference build for A/B
option('platform', type : 'combo',
       choices : ['native', 'winapi'],
       value : 'native',
       description : 'Platform backend: miniwin (native) or real DX via MinGW (winapi)')

# werror is a Meson built-in option (off by default) — see the comment in
# meson.build for the warning policy rationale.

# Directory containing the shipped LegoRR0.wad / LegoRR1.wad. When set, the
# resource-loader test (suite 'resource') also runs real-wad validation.
option('test_wads_dir', type : 'string', value : '',
       description : 'Directory with shipped wads for resource_test --real')

# AVI movie playback (finding 14 §4 / 23) is a *runtime* capability: the port
# dlopens the FFmpeg shared libraries at first use — no build-time link, no
# ffmpeg headers (LGPL-clean). This option controls how the build treats that
# dependency:
#   auto     (default) probe the runtime at configure time and report status;
#   enabled  require the runtime — configure fails if dlopen can't find it;
#   disabled compile the movie module out (stub) and drop the movie suite.
# The soname lists are owned by meson and passed to both the probe and the
# module as defines (single source of truth).
option('movies', type : 'feature',
       value : 'auto',
       description : 'AVI movie playback via dlopen of the FFmpeg runtime (auto/enabled/disabled)')
