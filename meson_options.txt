# Reconstruction source-tree root (relative to the project root).
# Relocating the tree is one option + moving the directory — no build rewiring.
option('src_root', type : 'string', value : 'src',
       description : 'Directory containing the reconstruction source tree')

# Platform backend that satisfies the DX6-era API surface (finding 15):
#   native : miniwin (SDL3) — primary, end-goal build, host compiler, no cross file
#   winapi : MinGW-w64 i686 vs real DirectX headers — reference build for A/B
option('platform', type : 'combo',
       choices : ['native', 'winapi'],
       value : 'native',
       description : 'Platform backend: miniwin (native) or real DX via MinGW (winapi)')

# `werror` is a Meson built-in option (off by default). Decompiled/reconstructed
# code is structurally noisy (opaque casts, dead stores); warnings are surfaced
# via -Wall -Wextra (warning_level=2) but errors opt-in: `-Dwerror=true`.
